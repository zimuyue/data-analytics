## 内置函数

### 1.1 数值函数

```sql
-- 四舍五入
SELECT round(5.76, 1);
SELECT round(5.7636, 2);
SELECT round(3.7636, -1); -- 0
SELECT round(5.7636, -1); -- 10
-- 截取位数
SELECT truncate(5.7345, 2); -- 5.73
SELECT truncate(5.7345, 0); -- 5
SELECT truncate(5.7345, -1); -- 0
SELECT truncate(5.7345, 5); -- 5.7345
-- 上限函数
-- 返回大于或等于参数的最小整数
SELECT ceiling(5.7); -- 6
SELECT ceiling(5.2); -- 6
-- 下限函数
SELECT floor(5.7); -- 5
-- 绝对值函数
SELECT abs(5.2); -- 5.2
SELECT abs(-5.2); -- -5.2
-- 随机数[0-1]
SELECT rand();
```
